window.exportLandingHtml = function() {
  if (!window._landingPageData) return;
  const html = `<!DOCTYPE html><html><head><meta charset="utf-8"><title>${_lsBrand()} — Landing</title><meta name="viewport" content="width=device-width,initial-scale=1"></head><body style="font-family:-apple-system,BlinkMacSystemFont,'Segoe UI',sans-serif;margin:0;color:#0F172A">${window._landingPageData.sections.map(s => s.html).join('')}</body></html>`;
  const blob = new Blob([html], { type:'text/html' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url; a.download = `${_lsDomain().replace(/\./g,'-')}-landing.html`; a.click();
  URL.revokeObjectURL(url);
  showToast('✅ Landing page exported');
};
